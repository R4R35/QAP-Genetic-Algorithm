cmake_minimum_required(VERSION 3.20)
project(CSSO_T3 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This project requires Windows.")
endif()
add_executable(t3 src/main.cpp src/common.cpp src/processes.cpp src/runner.cpp)
target_compile_features(t3 PRIVATE cxx_std_17)
target_compile_definitions(t3 PRIVATE UNICODE _UNICODE)
target_link_libraries(t3 PRIVATE advapi32)
add_executable(t3_child kit/t3_child.c)
target_compile_definitions(t3_child PRIVATE UNICODE _UNICODE)
target_link_libraries(t3_child PRIVATE psapi advapi32)
if(MSVC)
  target_compile_options(t3 PRIVATE /W4 /utf-8 /EHsc)
  target_compile_options(t3_child PRIVATE /W4 /utf-8)
endif()
